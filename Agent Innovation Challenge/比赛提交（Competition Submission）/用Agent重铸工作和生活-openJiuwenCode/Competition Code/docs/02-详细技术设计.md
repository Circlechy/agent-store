## 一、工具系统详细设计

### 1.1 工具基类接口

所有工具都基于 openJiuwen 的 `Tool` 基类：

```python
from openjiuwen.core.utils.tool.base import Tool
from openjiuwen.core.utils.tool.schema import ToolInfo
from openjiuwen.core.utils.tool.param import Param
from typing import Dict, Any, List
from pathlib import Path

class BaseCodeTool(Tool):
    """代码操作工具基类"""

    def __init__(self):
        super().__init__()
        self.name: str = ""
        self.description: str = ""
        self.params: List[Param] = []
        self.security_level: str = "safe"  # safe / dangerous / destructive

    async def ainvoke(self, inputs: Input, **kwargs) -> Output:
        """异步调用接口"""
        raise NotImplementedError

    def invoke(self, inputs: Input, **kwargs) -> Output:
        """同步调用接口"""
        raise NotImplementedError

    def get_tool_info(self) -> ToolInfo:
        """返回工具的元数据信息"""
        return ToolInfo(
            name=self.name,
            description=self.description,
            parameters=self._format_params()
        )

    def _format_params(self) -> Dict[str, Any]:
        """格式化参数为 Schema"""
        properties = {}
        required = []

        for param in self.params:
            properties[param.name] = {
                "type": param.type,
                "description": param.description
            }
            if param.default_value is not None:
                properties[param.name]["default"] = param.default_value
            if param.required:
                required.append(param.name)

        return {
            "type": "object",
            "properties": properties,
            "required": required
        }

    def validate_security(self, inputs: Dict[str, Any]) -> bool:
        """安全验证"""
        if self.security_level == "destructive":
            return self._validate_destructive_operation(inputs)
        return True

    def _validate_destructive_operation(self, inputs) -> bool:
        """验证破坏性操作"""
        # 需要用户确认
        return ask_user_confirmation(
            f"即将执行危险操作 {self.name}，确认继续？"
        )
```

```python
async def ainvoke(self, inputs: Input, **kwargs) -> Output:
    pattern = inputs.get("pattern")
    path = inputs.get("path", ".")

    # 构建命令
    cmd = ["rg", pattern, path]
    if not inputs.get("case_sensitive", False):
        cmd.append("-i")

    # 执行搜索
    process = await asyncio.create_subprocess_exec(
        *cmd,
        stdout=asyncio.subprocess.PIPE,
        stderr=asyncio.subprocess.PIPE
    )

    stdout, stderr = await process.communicate()

    # 格式化结果
    return Output(data=self._format_results(stdout.decode('utf-8')))
```

**安全检查：**

```python
def _is_command_safe(self, command: str) -> bool:
    """检查命令是否安全"""
    for blacklisted in self.blacklist:
        if blacklisted in command:
            return False
    return True

def _is_destructive(self, command: str) -> bool:
    """判断是否为破坏性操作"""
    destructive_patterns = [
        "rm -rf",
        "del ",
        "mkfs",
        "format"
    ]
    return any(pattern in command for pattern in destructive_patterns)
```

#### 核心设计

**Ref 引用系统**：通过 `browser_snapshot` 获取页面元素的引用标识（如 @e1, @e2），后续操作使用这些引用精确定位元素。

**会话管理**：自动绑定 jiuwen session_id，实现会话隔离。

**输出优化**：`browser_snapshot` 默认只返回可交互元素，减少 token 消耗。

#### 典型工作流

```
1. browser_open 打开目标网页
2. browser_snapshot 获取可交互元素列表
3. 根据引用执行 browser_click/browser_fill 等操作
4. 页面变化后重新获取快照
```

## 四、Agent 核心逻辑详细设计

### 4.1 主 Agent 类

```python
class OpenjiuwenCodeAgent(ReActAgent):
    """openjiuwen code 主 Agent"""

    def __init__(self, config: AgentConfig):
        super().__init__(config)

        # 组件初始化
        self.context_manager = ContextManager(config)
        self.reminder_generator = SystemReminderGenerator()
        self.tool_registry = ToolRegistry()

        # 模型配置
        self.main_model = config.main_model  # Sonnet 4
        self.small_model = config.small_model  # Haiku 3.5

    async def run(self, user_input: str) -> AsyncGenerator[AgentEvent, None]:
        """主运行循环"""
        # 1. 预处理
        await self._pre_process(user_input)

        # 2. 主题检测（可选）
        topic = await self._detect_topic(user_input)

        # 3. 构建消息
        messages = await self._build_messages(user_input)

        # 4. ReAct 循环
        async for event in self._react_loop(messages):
            yield event
```

### 4.2 ReAct 循环

```python
async def _react_loop(self, messages: List[LLMMessage]) -> AsyncGenerator[AgentEvent, None]:
    """ReAct 循环"""
    self.current_iteration = 0

    while self.current_iteration < self.max_iterations:
        # 1. 调用 LLM
        response = await self._call_llm(messages)

        # 2. 检查是否需要工具调用
        if not response.tool_calls:
            yield AgentEvent.content(response.content)
            break

        # 3. 执行工具调用
        for tool_call in response.tool_calls:
            yield AgentEvent.tool_call(tool_call)

            result = await self._execute_tool(tool_call)

            yield AgentEvent.tool_result(result)

            messages.append(ToolMessage(
                tool_call_id=tool_call.id,
                content=str(result)
            ))

        self.current_iteration += 1
```

```python
import yaml
from pathlib import Path
from typing import Dict, Any

class ConfigManager:
    """配置管理器"""

    def __init__(self, config_path: str = None):
        self.config_path = config_path or self._find_config()
        self.config = self._load_config()

    def _find_config(self) -> Path:
        """查找配置文件"""
        # 1. 当前目录
        if Path("config.yaml").exists():
            return Path("config.yaml")

        # 2. 用户目录
        user_config = Path.home() / ".openjiuwen-code" / "config.yaml"
        if user_config.exists():
            return user_config

        # 3. 默认配置
        return Path("config/config.example.yaml")

    def _load_config(self) -> Dict[str, Any]:
        """加载配置"""
        with open(self.config_path, 'r') as f:
            config = yaml.safe_load(f)

        # 环境变量替换
        config = self._substitute_env_vars(config)

        return config

    def _substitute_env_vars(self, config: Any) -> Any:
        """递归替换环境变量"""
        if isinstance(config, str):
            if config.startswith("${") and config.endswith("}"):
                var_name = config[2:-1]
                return os.getenv(var_name, config)
            return config

        elif isinstance(config, dict):
            return {
                k: self._substitute_env_vars(v)
                for k, v in config.items()
            }

        elif isinstance(config, list):
            return [
                self._substitute_env_vars(item)
                for item in config
            ]

        return config
```

```python
from concurrent import futures
import asyncio

class AgentServer:

    def __init__(self, port: int = 50051):
        self.port = port
        self.server = grpc.server(futures.ThreadPoolExecutor(max_workers=10))
        self.sessions: Dict[str, AgentSession] = {}

        add_AgentServiceServicer_to_server(
            agent_service, self.server
        )

        listen_addr = f"[::]:{self.port}"
        self.server.add_insecure_port(listen_addr)
        self.server.start()

        print(f"Agent Server running on port {self.port}")

        try:
            await self.server.wait_for_termination()

            await self.stop()

        """停止服务器"""
        await self.server.stop(0)

class AgentServiceImpl(AgentServiceServicer):
    """Agent 服务实现"""

    def __init__(self, sessions: Dict[str, AgentSession]):
        self.sessions = sessions

    async def CreateSession(
        self,
        request: CreateSessionRequest,
        context: grpc.ServicerContext
    ) -> SessionResponse:
        """创建新会话"""
        session_id = str(uuid.uuid4())

        # 创建 Agent 实例
        agent = OpenjiuwenCodeAgent(
            project_path=request.project_path,
            mode=AgentMode(request.initial_mode)
        )

        session = AgentSession(
            session_id=session_id,
            agent=agent,
            created_at=datetime.now()
        )

        self.sessions[session_id] = session

        return SessionResponse(
            session_id=session_id,
            status="created"
        )

    async def SendMessage(
        self,
        request_iterator: AsyncIterator[MessageRequest],
        context: grpc.ServicerContext
    ) -> AsyncIterator[MessageResponse]:
        """处理消息流"""
        # 获取会话
        first_request = await request_iterator.__anext__()
        session = self.sessions.get(first_request.session_id)

        if not session:
            context.set_code(grpc.StatusCode.NOT_FOUND)
            return

        # 运行 Agent
        async for event in session.agent.run(first_request.content):
            yield MessageResponse(
                event_type=MessageType.THINKING,
                data=json.dumps(event),
                timestamp=int(time.time())
            )
```

```python
        content=content,
        uri=uri,
        mode="review"
    )

    # 转换为 LSP 诊断格式
    diagnostics = []
    for issue in result.get("issues", []):
        diagnostics.append(Diagnostic(
            range=Range(
                start=Position(
                    line=issue["line"] - 1,
                    character=issue["column"]
                ),
                end=Position(
                    line=issue["end_line"] - 1,
                    character=issue["end_column"]
                )
            ),
            message=issue["message"],
            severity=DiagnosticSeverity.Warning if issue["level"] == "warning" else DiagnosticSeverity.Error,
            source="opencode"
        ))

    return diagnostics
```

```python
# packages/server/agents/lsp_integration.py

class LSPIntegratedAgent(ReActAgent):
    """集成 LSP 的 Agent"""

    def __init__(self, *args, **kwargs):
        super().__init__(*args, **kwargs)
        self.lsp_client = LSPClient()

    async def get_code_diagnostics(self, file_path: str) -> List[Dict]:
        """获取代码诊断信息"""
        return await self.lsp_client.get_diagnostics(file_path)

    async def get_code_symbols(self, file_path: str) -> List[Dict]:
        """获取代码符号"""
        return await self.lsp_client.get_symbols(file_path)

    async def review_mode_analysis(self, file_path: str) -> Dict:
        """review 模式下的增强分析"""
        # 1. 获取 LSP 诊断
        diagnostics = await self.get_code_diagnostics(file_path)

        # 2. 读取文件内容
        content = await self.read_file(file_path)

        # 3. 结合诊断信息进行分析
        analysis = await self.analyze_with_diagnostics(
            content=content,
            diagnostics=diagnostics
        )

        return analysis
```
