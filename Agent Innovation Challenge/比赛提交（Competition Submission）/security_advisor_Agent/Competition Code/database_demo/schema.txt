-- -------------------------------
-- 2. Tables
-- -------------------------------

CREATE TABLE IF NOT EXISTS public.customers (
    customer_id   UUID PRIMARY KEY,
    email         TEXT,
    full_name     TEXT,
    phone         TEXT,
    country_code  CHAR(2),
    created_at    TIMESTAMP
);

CREATE TABLE IF NOT EXISTS public.auth_accounts (
    customer_id    UUID REFERENCES public.customers(customer_id),
    password_hash  TEXT,
    mfa_secret     TEXT,
    last_login     TIMESTAMP
);

CREATE TABLE IF NOT EXISTS public.payments (
    payment_id   UUID PRIMARY KEY,
    customer_id  UUID REFERENCES public.customers(customer_id),
    card_last4   CHAR(4),
    card_token   TEXT,
    amount       NUMERIC,
    currency     CHAR(3),
    created_at   TIMESTAMP
);

CREATE TABLE IF NOT EXISTS public.support_tickets (
    ticket_id    UUID PRIMARY KEY,
    customer_id  UUID,
    message      TEXT,
    created_at   TIMESTAMP,
    handled_by   TEXT
);

CREATE TABLE IF NOT EXISTS public.usage_events (
    event_id     UUID PRIMARY KEY,
    customer_id  UUID REFERENCES public.customers(customer_id),
    event_type   TEXT,
    event_time   TIMESTAMP
);

-- -------------------------------
-- 3. Resource Labels (taxonomy)
-- -------------------------------

/* Direct identifiers (GDPR Art. 4) */
CREATE RESOURCE LABEL IF NOT EXISTS lbl_pii_raw
ADD COLUMN (
    public.customers.email,
    public.customers.full_name,
    public.customers.phone
);

/* Low sensitivity personal data */
CREATE RESOURCE LABEL IF NOT EXISTS lbl_pii_low
ADD COLUMN (
    public.customers.country_code
);

/* Authentication secrets */
CREATE RESOURCE LABEL IF NOT EXISTS lbl_auth_secret
ADD COLUMN (
    public.auth_accounts.password_hash,
    public.auth_accounts.mfa_secret
);

/* PCI-like payment data */
CREATE RESOURCE LABEL IF NOT EXISTS lbl_pci
ADD COLUMN (
    public.payments.card_token
);

/* Free-text user content (high GDPR risk) */
CREATE RESOURCE LABEL IF NOT EXISTS lbl_free_text_pii
ADD COLUMN (
    public.support_tickets.message
);

-- -------------------------------
-- 4. Optional table-level labels
-- -------------------------------

/* Customer-provided content */
CREATE RESOURCE LABEL IF NOT EXISTS lbl_customer_content
ADD TABLE (
    public.support_tickets
);

/* Behavioral / analytics data */
CREATE RESOURCE LABEL IF NOT EXISTS lbl_behavioral
ADD TABLE (
    public.usage_events
);
